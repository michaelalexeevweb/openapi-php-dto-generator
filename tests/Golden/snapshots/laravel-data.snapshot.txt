                    $validator->errors()->add($foundPath, sprintf('%s expects object, got array.', $foundPath));
                }
            }
        }
    }

    /**
     * Every node a (possibly wildcarded) path reaches, keyed by the concrete path that got there.
     *
     * @param array<int, string> $segments
     * @return array<string, mixed>
     */
    private static function rawNodesAt(mixed $node, array $segments, string $path): array
    {
        if ($segments === []) {
            return [$path => $node];
        }

        $segment = array_shift($segments);
        if ($segment === '*') {
            // `*` stands for both an array index and a map key, so a JSON object node is walked too.
            if ($node instanceof stdClass) {
                $node = get_object_vars($node);
            }
            if (!is_array($node)) {
                return [];
            }

            $found = [];
            foreach ($node as $index => $item) {
                $itemPath = $path === '' ? (string)$index : $path . '.' . $index;
                foreach (self::rawNodesAt($item, $segments, $itemPath) as $deepPath => $deepNode) {
                    $found[$deepPath] = $deepNode;
                }
            }

            return $found;
        }

        if (!$node instanceof stdClass || !property_exists($node, $segment)) {
            return [];
        }

        return self::rawNodesAt($node->{$segment}, $segments, $path === '' ? $segment : $path . '.' . $segment);
    }

    /**
     * @param array<string, mixed> $schema
     * @return array<int, string>
     */
    private static function validateOpenApiNode(mixed $value, array $schema, string $path, int $depth): array
    {
        if ($depth >= self::OPENAPI_MAX_VALIDATION_DEPTH) {
            return [];
        }

        $errors = [];
        $structuredValue = self::normalizeOpenApiStructuralValue($value);

        if (is_array($structuredValue) && is_array($schema['required'] ?? null)) {
            foreach ($schema['required'] as $requiredProp) {
                if (is_string($requiredProp) && !array_key_exists($requiredProp, $structuredValue)) {
                    $errors[] = sprintf('%s is required', self::openApiChildPath($path, $requiredProp));
                }
            }
        }

        $definedProperties = [];
        $patternMatchedProperties = [];

        if (is_array($structuredValue)) {
            $extraPropertySchema = $schema['additionalProperties'] ?? null;
            $unevaluatedProperties = $schema['unevaluatedProperties'] ?? null;
            $evaluatedProperties = [...$definedProperties, ...$patternMatchedProperties];

            if ($extraPropertySchema === false || is_array($extraPropertySchema)) {
                foreach ($structuredValue as $propertyName => $propertyValue) {
                    if (!is_string($propertyName) || array_key_exists($propertyName, $evaluatedProperties)) {
                        continue;
                    }
                    if ($extraPropertySchema === false) {
                        $errors[] = sprintf('%s has additional property "%s" which is not allowed', $path, $propertyName);
                        continue;
                    }
                    $errors = [
                        ...$errors,
                        ...self::validateOpenApiNode($propertyValue, $extraPropertySchema, self::openApiChildPath($path, $propertyName), $depth + 1),
                    ];
                    $evaluatedProperties[$propertyName] = true;
                }
            }
        }

        return $errors;
    }

    private static function normalizeOpenApiStructuralValue(mixed $value): mixed
    {
        if (is_array($value) || !is_object($value)) {
            return $value;
        }

        // Any other object: only its public state is visible, under PHP property names.
        return array_filter(
            get_object_vars($value),
            static fn(mixed $propertyValue): bool => $propertyValue !== null,
        );
    }

    private static function openApiChildPath(string $path, string $propertyName): string
    {
        // The first level names the field the way the runtime validator's caller does, so both
        // modes report the same subject: `field "tags" must ...`.
        return $path === 'payload'
            ? sprintf('field "%s"', $propertyName)
            : $path . '.' . $propertyName;
    }
}

                $errors = [
                    ...$errors,
                    ...self::validateOpenApiNode($itemValue, $schema['items'], $path . '[' . $index . ']', $depth + 1),

        return $errors;

        $container = Container::getInstance();
        if (!$container->bound('request')) {
            return null;
        }

        $request = $container->make('request');

        return $request instanceof Request ? $request->getContent() : null;

    /**
     * Adds an error for every `type: object` path the raw body sent as a JSON array.
     */
    private static function assertRawObjectShapes(Validator $validator, string $rawJson): void
    {
        // Decoded WITHOUT the assoc flag on purpose: that is what keeps a JSON object apart from a
        // JSON array. Laravel's own `$validator->getData()` cannot answer this question.
        $decoded = json_decode($rawJson, false);
        if (!$decoded instanceof stdClass) {
            return;
        }

        foreach (self::OPENAPI_OBJECT_PATHS as $path) {
            foreach (self::rawNodesAt($decoded, explode('.', $path), '') as $foundPath => $node) {
                if (is_array($node)) {
                    $validator->errors()->add($foundPath, sprintf('%s expects object, got array.', $foundPath));
                }
            }
        }
    }

    /**
     * Every node a (possibly wildcarded) path reaches, keyed by the concrete path that got there.
     *
     * @param array<int, string> $segments
     * @return array<string, mixed>
     */
    private static function rawNodesAt(mixed $node, array $segments, string $path): array
    {
        if ($segments === []) {
            return [$path => $node];
        }

        $segment = array_shift($segments);
        if ($segment === '*') {
            // `*` stands for both an array index and a map key, so a JSON object node is walked too.
            if ($node instanceof stdClass) {
                $node = get_object_vars($node);
            }
            if (!is_array($node)) {
                return [];
            }

            $found = [];
            foreach ($node as $index => $item) {
                $itemPath = $path === '' ? (string)$index : $path . '.' . $index;
                foreach (self::rawNodesAt($item, $segments, $itemPath) as $deepPath => $deepNode) {
                    $found[$deepPath] = $deepNode;
                }
            }

            return $found;
        }

        if (!$node instanceof stdClass || !property_exists($node, $segment)) {
            return [];
        }

        return self::rawNodesAt($node->{$segment}, $segments, $path === '' ? $segment : $path . '.' . $segment);
    }

    /**
     * @param array<string, mixed> $schema
     * @return array<int, string>
     */
    private static function validateOpenApiNode(mixed $value, array $schema, string $path, int $depth): array
    {
        if ($depth >= self::OPENAPI_MAX_VALIDATION_DEPTH) {
            return [];
        }

        $errors = [];
        if ($value === null && ($schema['nullable'] ?? false) === true) {
            return [];
        }

        $structuredValue = self::normalizeOpenApiStructuralValue($value);

        if (is_array($structuredValue) && is_array($schema['required'] ?? null)) {
            foreach ($schema['required'] as $requiredProp) {
                if (is_string($requiredProp) && !array_key_exists($requiredProp, $structuredValue)) {
                    $errors[] = sprintf('%s is required', self::openApiChildPath($path, $requiredProp));
                }
            }
        }

        $listValue = is_array($value) && array_is_list($value);

        if ($listValue && is_array($schema['items'] ?? null)) {
            foreach ($value as $index => $itemValue) {
                $errors = [
                    ...$errors,
                    ...self::validateOpenApiNode($itemValue, $schema['items'], $path . '[' . $index . ']', $depth + 1),
                ];
            }
        }

        return $errors;
    }

    private static function normalizeOpenApiStructuralValue(mixed $value): mixed
    {
        if (is_array($value) || !is_object($value)) {
            return $value;
        }

        // Any other object: only its public state is visible, under PHP property names.
        return array_filter(
            get_object_vars($value),
            static fn(mixed $propertyValue): bool => $propertyValue !== null,
        );
    }

    private static function openApiChildPath(string $path, string $propertyName): string
    {
        // The first level names the field the way the runtime validator's caller does, so both
        // modes report the same subject: `field "tags" must ...`.
        return $path === 'payload'
            ? sprintf('field "%s"', $propertyName)
            : $path . '.' . $propertyName;
    }

     * Every branch was gated out by its declared type, so no branch has a reason to report. Naming the
     * types the union does accept turns an unactionable sentence into one the caller can act on. Same
     * wording as the runtime validator, so a payload reads the same whichever mode answered it.
     *
     * @param array<int, mixed> $branches
     */
    private static function describeOpenApiUnionMismatch(string $path, string $kind, array $branches, mixed $value): string
    {
        $types = [];
        foreach ($branches as $branch) {
            if (!is_array($branch)) {
                continue;
            }

            $declared = is_array($branch['type'] ?? null) ? $branch['type'] : [$branch['type'] ?? null];
            foreach ($declared as $type) {
                if (is_string($type) && $type !== '' && !in_array($type, $types, true)) {
                    $types[] = $type;
                }
            }
        }

        if ($types === []) {
            return sprintf('%s does not match any %s branch', $path, $kind);
        }

        $last = array_pop($types);
        $expected = $types === [] ? $last : implode(', ', $types) . ' or ' . $last;

        return sprintf(
            '%s does not match any %s branch (expected %s, got %s)',
            $path,
            $kind,
            $expected,
            match (true) {
                $value === null => 'null',
                is_bool($value) => 'boolean',
                is_int($value) => 'integer',
                is_float($value) => 'number',
                is_string($value) => 'string',
                is_array($value) => array_is_list($value) ? 'array' : 'object',
                default => 'object',
            },
        );
    }

    private static function matchesOpenApiCallbackType(mixed $value, mixed $type): bool
    {
        if (!is_string($type) || $type === '') {
            return true;
        }

        return match ($type) {
            'integer' => is_int($value) || (is_float($value) && is_finite($value) && floor($value) === $value),
            'string' => is_string($value),
            default => true,
        };
    }

    private static function normalizeOpenApiStructuralValue(mixed $value): mixed
    {
        if (is_array($value) || !is_object($value)) {
            return $value;
        }

        // Any other object: only its public state is visible, under PHP property names.
        return array_filter(
            get_object_vars($value),
            static fn(mixed $propertyValue): bool => $propertyValue !== null,
        );
    }

    /**
     * @param array<int, string> $keys
     */
    private static function schemaHasAnyOpenApiKey(array $schema, array $keys): bool
    {
        foreach ($keys as $key) {
            if (array_key_exists($key, $schema)) {
                return true;
            }
        }

        return false;
    }

    private static function toFloatConstraint(mixed $value): ?float
    {
        return is_int($value) || is_float($value) ? (float)$value : null;
    }

    private static function stringifyOpenApiNumber(float $value): string
    {
        $rendered = json_encode($value);
        return is_string($rendered) ? $rendered : (string)$value;
    }

    private static function isValidOpenApiStringFormat(string $value, string $format): bool
    {
        return match ($format) {
            'uuid' => self::isValidOpenApiUuid($value),
            'uri-template' => self::isValidOpenApiUriTemplate($value),
            default => true,
        };
    }

    private static function isValidOpenApiUuid(string $value): bool
    {
        return preg_match('/^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[1-8][0-9a-fA-F]{3}-[89abAB][0-9a-fA-F]{3}-[0-9a-fA-F]{12}$/', $value) === 1;
    }

    private static function isValidOpenApiUriReference(string $value): bool
    {
        return preg_match('/[\s\x00-\x1F\x7F]/u', $value) !== 1;
    }

    private static function isValidOpenApiUriTemplate(string $value): bool
    {
        // RFC 6570: a URI-reference that may embed {expression} blocks. Every brace has to belong to a
        // well-formed expression — an optional operator followed by a comma-separated list of varspecs,
        // each a varname with an optional ":" prefix-length or "*" explode modifier. Strip the valid
        // expressions; a surviving brace means the template is malformed.
        if (!self::isValidOpenApiUriReference($value)) {
            return false;
        }

        if (!str_contains($value, '{')) {
            return !str_contains($value, '}');
        }

        $expression = '\{[+#./;?&=,!@|]?(?:[\p{L}\p{N}_%]+(?::[1-9][0-9]{0,3}|\*)?)'
            . '(?:,[\p{L}\p{N}_%]+(?::[1-9][0-9]{0,3}|\*)?)*\}';
        $stripped = preg_replace('~' . $expression . '~u', '', $value);

        return $stripped !== null && !str_contains($stripped, '{') && !str_contains($stripped, '}');
    }

        foreach ($branches as $branch) {
            if (!is_array($branch)) {
                continue;
            }

            $declared = is_array($branch['type'] ?? null) ? $branch['type'] : [$branch['type'] ?? null];
            foreach ($declared as $type) {
                if (is_string($type) && $type !== '' && !in_array($type, $types, true)) {
                    $types[] = $type;
                }
            }
        }

        if ($types === []) {
            return sprintf('%s does not match any %s branch', $path, $kind);
        }

        $last = array_pop($types);
        $expected = $types === [] ? $last : implode(', ', $types) . ' or ' . $last;

        return sprintf(
            '%s does not match any %s branch (expected %s, got %s)',
            $path,
            $kind,
            $expected,
            match (true) {
                $value === null => 'null',
                is_bool($value) => 'boolean',
                is_int($value) => 'integer',
                is_float($value) => 'number',
                is_string($value) => 'string',
                is_array($value) => array_is_list($value) ? 'array' : 'object',
                default => 'object',
            },
        );
    }

    private static function matchesOpenApiCallbackType(mixed $value, mixed $type): bool
    {
        if (!is_string($type) || $type === '') {
            return true;
        }

        return match ($type) {
            'integer' => is_int($value) || (is_float($value) && is_finite($value) && floor($value) === $value),
            'string' => is_string($value),
            'object' => is_array($value) || is_object($value),
            default => true,
        };
    }

    private static function normalizeOpenApiStructuralValue(mixed $value): mixed
    {
        if (is_array($value) || !is_object($value)) {
            return $value;
        }

        // Any other object: only its public state is visible, under PHP property names.
        return array_filter(
            get_object_vars($value),
            static fn(mixed $propertyValue): bool => $propertyValue !== null,
        );
    }

    private static function openApiChildPath(string $path, string $propertyName): string
    {
        // The first level names the field the way the runtime validator's caller does, so both
        // modes report the same subject: `field "tags" must ...`.
        return $path === 'payload'
            ? sprintf('field "%s"', $propertyName)
            : $path . '.' . $propertyName;
    }

    /**
     * @param array<int, string> $keys
     */
    private static function schemaHasAnyOpenApiKey(array $schema, array $keys): bool
    {
        foreach ($keys as $key) {
            if (array_key_exists($key, $schema)) {
                return true;
            }
        }

        return false;
    }

    private static function toIntConstraint(mixed $value): ?int
    {
        if (is_int($value)) {
            return $value;
        }
        if (is_float($value) && floor($value) === $value) {
            return (int)$value;
        }

        return null;
    }

    /**
     * The half of the schema Laravel's rule vocabulary cannot express — composition, conditionals,
     * `contains`, `unevaluated*`, `content*`, `propertyNames`, `discriminator` — checked against the
     * raw payload the validator already holds.
     *
     * laravel-data calls this itself, and only on the ROOT object: a nested `Data` class never gets its
     * own `withValidator()` run, which is why every check above is flattened to a path from the root.
     *
     * One check needs the UNDECODED body, because `type: object` versus `type: array` is a property of
     * the wire shape that PHP loses — a JSON object keyed 0..n-1 and a JSON array decode to the same
     * list. Only the request still has those bytes, so on a `validateAndCreate($array)` call, where
     * there is no request, that single check is skipped and everything else still runs.
     */
    public static function withValidator(Validator $validator): void
    {
        $validator->after(static function (Validator $validator): void {
            /** @var array<string, mixed> $payload */
            $payload = $validator->getData();

            foreach (self::OPENAPI_VALIDATION_CONSTRAINTS as $property => $schema) {
                if (!array_key_exists($property, $payload)) {
                    continue;
                }

                foreach (self::validateOpenApiNode($payload[$property], $schema, $property, 0) as $error) {
                    $validator->errors()->add($property, str_ends_with($error, '.') ? $error : $error . '.');
                }
            }

            $rawJson = self::rawRequestBody();
            if ($rawJson !== null && $rawJson !== '') {
                self::assertRawObjectShapes($validator, $rawJson);
            }
        });
    }

    /**

    /**
     * Adds an error for every `type: object` path the raw body sent as a JSON array.
     */
    private static function assertRawObjectShapes(Validator $validator, string $rawJson): void
    {
        // Decoded WITHOUT the assoc flag on purpose: that is what keeps a JSON object apart from a
        // JSON array. Laravel's own `$validator->getData()` cannot answer this question.
        $decoded = json_decode($rawJson, false);
        if (!$decoded instanceof stdClass) {
            return;
        }

        foreach (self::OPENAPI_OBJECT_PATHS as $path) {
            foreach (self::rawNodesAt($decoded, explode('.', $path), '') as $foundPath => $node) {
                if (is_array($node)) {
                    $validator->errors()->add($foundPath, sprintf('%s expects object, got array.', $foundPath));
                }
            }
        }
    }

    /**
     * Every node a (possibly wildcarded) path reaches, keyed by the concrete path that got there.
     *
     * @param array<int, string> $segments
     * @return array<string, mixed>
     */
    private static function rawNodesAt(mixed $node, array $segments, string $path): array
    {
        if ($segments === []) {
            return [$path => $node];
        }

        $segment = array_shift($segments);
        if ($segment === '*') {
            // `*` stands for both an array index and a map key, so a JSON object node is walked too.
            if ($node instanceof stdClass) {
                $node = get_object_vars($node);
            }
            if (!is_array($node)) {
                return [];
            }

            $found = [];
            foreach ($node as $index => $item) {
                $itemPath = $path === '' ? (string)$index : $path . '.' . $index;
                foreach (self::rawNodesAt($item, $segments, $itemPath) as $deepPath => $deepNode) {
                    $found[$deepPath] = $deepNode;
                }
            }

            return $found;
        }

        if (!$node instanceof stdClass || !property_exists($node, $segment)) {
            return [];
        }

        return self::rawNodesAt($node->{$segment}, $segments, $path === '' ? $segment : $path . '.' . $segment);
    }

    /**
     * @param array<string, mixed> $schema
     * @return array<int, string>
     */
    private static function validateOpenApiNode(mixed $value, array $schema, string $path, int $depth): array
    {
        if ($depth >= self::OPENAPI_MAX_VALIDATION_DEPTH) {
            return [];
        }

        $errors = [];
        $normalizedValue = $value;
        $structuredValue = self::normalizeOpenApiStructuralValue($value);

        if (is_array($schema['oneOf'] ?? null)) {
            $matchingBranches = 0;
            $branchErrors = [];
            foreach ($schema['oneOf'] as $branch) {
                if (!is_array($branch)) {
                    continue;
                }

                if (is_string($branchClass = $branch['x-php-instanceof'] ?? null) && $branchClass !== '') {
                    $branchFqcn = __NAMESPACE__ . '\\' . ltrim($branchClass, '\\');
                    if (is_object($value) && $value instanceof $branchFqcn) {
                        $matchingBranches++;
                    }
                    continue;
                }

                if (array_key_exists('type', $branch) && !self::matchesOpenApiCallbackType($normalizedValue, $branch['type'])) {
                    continue;
                }
                // Keep why each branch failed: with no match at all those reasons are far more
                // useful than "does not match any oneOf branch".
                $errorsForBranch = self::validateOpenApiNode($value, $branch, $path, $depth + 1);
                if ($errorsForBranch === []) {
                    $matchingBranches++;
                    continue;
                }

                $branchErrors = [...$branchErrors, ...$errorsForBranch];
            }

            if ($matchingBranches > 1) {
                $errors[] = sprintf('%s matches more than one allowed oneOf branch', $path);
            } elseif ($matchingBranches === 0) {
                $errors = $branchErrors === []
                    ? [...$errors, self::describeOpenApiUnionMismatch($path, 'oneOf', $schema['oneOf'], $value)]
                    : [...$errors, ...array_values(array_unique($branchErrors))];
            }

            if ($matchingBranches === 1) {
                $discriminatorProperty = is_string($schema['x-discriminator-property'] ?? null)
                    ? $schema['x-discriminator-property']
                    : null;
                $discriminatorPhpProperty = is_string($schema['x-discriminator-php-property'] ?? null)
                    ? $schema['x-discriminator-php-property']
                    : null;
                $discriminatorMap = is_array($schema['x-discriminator-map'] ?? null)
                    ? $schema['x-discriminator-map']
                    : null;

                if ($discriminatorProperty !== null && $discriminatorMap !== null && is_object($value)) {
                    $phpProperty = $discriminatorPhpProperty ?? $discriminatorProperty;
                    // The DTO keeps its state private: read the discriminator from the payload it
                    // reports, keyed by the OpenAPI name, and fall back to a public property for a
                    // hand-written object.
                    $discriminatorPayload = self::normalizeOpenApiStructuralValue($value);
                    $discriminatorCarrier = is_array($discriminatorPayload) ? $discriminatorPayload : [];
                    if (!array_key_exists($discriminatorProperty, $discriminatorCarrier) && !property_exists($value, $phpProperty)) {
                        $errors[] = sprintf('%s discriminator property %s is required', $path, $discriminatorProperty);
                    } else {
                        $discriminatorValue = $discriminatorCarrier[$discriminatorProperty]
                            ?? (property_exists($value, $phpProperty) ? $value->{$phpProperty} : null);
                        $mapKey = is_scalar($discriminatorValue) ? (string)$discriminatorValue : null;
                        if ($mapKey === null || !array_key_exists($mapKey, $discriminatorMap)) {
                            $errors[] = sprintf(
                                '%s discriminator %s must be one of: %s',
                                $path,
                                $discriminatorProperty,
                                implode(', ', array_keys($discriminatorMap)),
                            );
                        } elseif (is_string($expectedClass = $discriminatorMap[$mapKey]) && $expectedClass !== '') {
                            $expectedFqcn = __NAMESPACE__ . '\\' . ltrim($expectedClass, '\\');
                            if (!$value instanceof $expectedFqcn) {
                                $errors[] = sprintf('%s discriminator %s must match concrete class', $path, $discriminatorProperty);
                            }
                        }
                    }
                }
            }
        }

        if (array_key_exists('type', $schema)) {
            $typeConstraint = $schema['type'];
            if (is_string($typeConstraint)) {
                if (!self::matchesOpenApiCallbackType($normalizedValue, $typeConstraint)) {
                    $errors[] = $typeConstraint === 'array' && is_array($normalizedValue)
                        ? sprintf('%s must be a JSON array (list with sequential keys), got an associative array', $path)
                        : sprintf('%s must be of type %s', $path, $typeConstraint);
                }
            }
        }

        if (is_string($normalizedValue) && self::schemaHasAnyOpenApiKey($schema, ['minLength'])) {
            $length = mb_strlen($normalizedValue);
            $minLength = self::toIntConstraint($schema['minLength'] ?? null);
            if ($minLength !== null && $length < $minLength) {
                $errors[] = sprintf('%s length must be at least %d characters', $path, $minLength);
            }
        }

        if (is_array($structuredValue) && is_array($schema['required'] ?? null)) {
            foreach ($schema['required'] as $requiredProp) {
                if (is_string($requiredProp) && !array_key_exists($requiredProp, $structuredValue)) {
                    $errors[] = sprintf('%s is required', self::openApiChildPath($path, $requiredProp));
                }
            }
        }

        $definedProperties = [];
        $patternMatchedProperties = [];

        if (is_array($schema['properties'] ?? null) && is_array($structuredValue)) {
            foreach ($schema['properties'] as $propertyName => $propertySchema) {
                if (!is_string($propertyName) || !is_array($propertySchema)) {
                    continue;
                }
                $definedProperties[$propertyName] = true;
                if (!array_key_exists($propertyName, $structuredValue)) {
                    continue;
                }
                $errors = [
                    ...$errors,
                    ...self::validateOpenApiNode($structuredValue[$propertyName], $propertySchema, self::openApiChildPath($path, $propertyName), $depth + 1),
                ];
            }
        }

        $listValue = is_array($value) && array_is_list($value);

        if ($listValue && is_array($schema['items'] ?? null)) {
            foreach ($value as $index => $itemValue) {
                $errors = [
                    ...$errors,
                    ...self::validateOpenApiNode($itemValue, $schema['items'], $path . '[' . $index . ']', $depth + 1),
                ];
            }
        }

        if (is_array($schema['allOf'] ?? null)) {
            foreach ($schema['allOf'] as $branch) {
                if (!is_array($branch)) {
                    continue;
                }

                $errors = [...$errors, ...self::validateOpenApiNode($value, $branch, $path, $depth + 1)];
            }
        }

        if (is_array($schema['anyOf'] ?? null)) {
            $anyOfMatched = false;
            $anyOfErrors = [];
            foreach ($schema['anyOf'] as $branch) {
                if (!is_array($branch)) {
                    continue;
                }

                if (array_key_exists('type', $branch) && !self::matchesOpenApiCallbackType($normalizedValue, $branch['type'])) {
                    continue;
                }
                $errorsForBranch = self::validateOpenApiNode($value, $branch, $path, $depth + 1);
                if ($errorsForBranch === []) {
                    $anyOfMatched = true;
                    break;
                }

                $anyOfErrors = [...$anyOfErrors, ...$errorsForBranch];
            }

            if (!$anyOfMatched) {
                $errors = $anyOfErrors === []
                    ? [...$errors, self::describeOpenApiUnionMismatch($path, 'anyOf', $schema['anyOf'], $value)]
                    : [...$errors, ...array_values(array_unique($anyOfErrors))];
            }
        }

        if (is_array($structuredValue)) {
            $extraPropertySchema = $schema['additionalProperties'] ?? null;
            $unevaluatedProperties = $schema['unevaluatedProperties'] ?? null;
            $evaluatedProperties = [...$definedProperties, ...$patternMatchedProperties];

            if ($extraPropertySchema === false || is_array($extraPropertySchema)) {
                foreach ($structuredValue as $propertyName => $propertyValue) {
                    if (!is_string($propertyName) || array_key_exists($propertyName, $evaluatedProperties)) {
                        continue;
                    }
                    if ($extraPropertySchema === false) {
                        $errors[] = sprintf('%s has additional property "%s" which is not allowed', $path, $propertyName);
                        continue;
                    }
                    $errors = [
                        ...$errors,
                        ...self::validateOpenApiNode($propertyValue, $extraPropertySchema, self::openApiChildPath($path, $propertyName), $depth + 1),
                    ];
                    $evaluatedProperties[$propertyName] = true;
                }
            }
        }

        return $errors;
    }

    /**
     * @param array<string, mixed> $schema
     */
    private static function matchesOpenApiNode(mixed $value, array $schema, int $depth): bool
    {
        return self::validateOpenApiNode($value, $schema, 'payload', $depth) === [];
    }

    /**
     * Every branch was gated out by its declared type, so no branch has a reason to report. Naming the
     * types the union does accept turns an unactionable sentence into one the caller can act on. Same
     * wording as the runtime validator, so a payload reads the same whichever mode answered it.
     *
     * @param array<int, mixed> $branches
     */
    private static function describeOpenApiUnionMismatch(string $path, string $kind, array $branches, mixed $value): string
    {
        $types = [];
        foreach ($branches as $branch) {
            if (!is_array($branch)) {
                continue;
            }

            $declared = is_array($branch['type'] ?? null) ? $branch['type'] : [$branch['type'] ?? null];
            foreach ($declared as $type) {
                if (is_string($type) && $type !== '' && !in_array($type, $types, true)) {
                    $types[] = $type;
                }
            }
        }

        if ($types === []) {
            return sprintf('%s does not match any %s branch', $path, $kind);
        }

        $last = array_pop($types);
        $expected = $types === [] ? $last : implode(', ', $types) . ' or ' . $last;

        return sprintf(
            '%s does not match any %s branch (expected %s, got %s)',
            $path,
            $kind,
            $expected,
            match (true) {
                $value === null => 'null',
                is_bool($value) => 'boolean',
                is_int($value) => 'integer',
                is_float($value) => 'number',
                is_string($value) => 'string',
                is_array($value) => array_is_list($value) ? 'array' : 'object',
                default => 'object',
            },
        );
    }

    private static function matchesOpenApiCallbackType(mixed $value, mixed $type): bool
    {
        if (!is_string($type) || $type === '') {
            return true;
        }

        return match ($type) {
            'integer' => is_int($value) || (is_float($value) && is_finite($value) && floor($value) === $value),
